package com.sanwi22;

import java.util.ArrayList;
import java.util.List;

public class Rook extends Piece{

    public Rook(Position position, Boolean isWhite) {
        super(position, isWhite);
    }

    @Override
    public List<Position> getLegalMoves(Board board) {

        ArrayList<Position> legalPos = new ArrayList<>();
        int row = getPosition().getRow();
        int col = getPosition().getCol();
        Position position = new Position(row, col);
        String[] colLetters = {"a", "b", "c", "d", "e", "f", "g", "h"};

        if(this.isWhite()) {
            for(int i = row +1; i < 8; i++) {
                position.setRow(i);
                try {
                    if((board.getPiece(position) == null) || (!board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[col] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }

            position.setRow(row);
            for(int i = row -1; i >= 0; i--) {
                position.setRow(i);
                try {
                    if((board.getPiece(position) == null) || (!board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[col] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }

            position.setRow(row);
            for(int i = col+1; i <=7; i++) {
                position.setCol(i);
                //System.out.println(position.getCol());
                try {
                    if((board.getPiece(position) == null) || (board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[position.getCol()] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }

            position.setCol(col);
            for(int i = col-1; i >= 0; i--) {
                position.setCol(i);
                //System.out.println(position.getCol());
                try {
                    if ((board.getPiece(position) == null) || (board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[position.getCol()] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }
        } else {
            for(int i = row +1; i < 8; i++) {
                position.setRow(i);
                try {
                    if((board.getPiece(position) == null) || (!board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[col] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }

            position.setRow(row);
            for(int i = row -1; i >= 0; i--) {
                position.setRow(i);
                try {
                    if((board.getPiece(position) == null) || (board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[col] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }

            position.setRow(row);
            for(int i = col+1; i <=7; i++) {
                position.setCol(i);
                //System.out.println(position.getCol());
                try {
                    if((board.getPiece(position) == null) || (board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[position.getCol()] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }

            position.setCol(col);
            for(int i = col-1; i >= 0; i--) {
                position.setCol(i);
                //System.out.println(position.getCol());
                try {
                    if ((board.getPiece(position) == null) || (board.getPiece(position).isWhite())) {
                        legalPos.add(position);
                        System.out.println(colLetters[position.getCol()] + (position.getRow() + 1));
                    } else {
                        break;
                    }
                } catch (ArrayIndexOutOfBoundsException ignored) {}
            }
        }
        return List.of();
    }

    public static void main(String[] args) {
        Board board = new Board();
        Pawn pawn = new Pawn(new Position(6, 7), true);
        board.placePiece(pawn);
        Rook rook = new Rook(new Position(7, 7), true);
        board.placePiece(rook);
        rook.getLegalMoves(board);
    }
}
